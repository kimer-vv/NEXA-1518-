function askUseSavedForm(){
 return new Promise(resolve=>{
   const box=$('savedFormPrompt');
   const yes=$('useSavedForm');
   const no=$('keepSavedInactive');
   let done=false;
   const finish=value=>{
     if(done)return;
     done=true;
     box.classList.remove('open');
     yes.onclick=null;
     no.onclick=null;
     resolve(value);
   };
   yes.onclick=()=>finish(true);
   no.onclick=()=>finish(false);
   box.classList.add('open');
 });
}
